# ADBuddy M3.0 · Protocolo común · Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** dejar escrito y probado a los dos lados el contrato completo del protocolo app ↔ host de M3, M4 y M5 (tipos, fixtures compartidas, clases de mensaje, `rid` y `error`, tramas binarias) y los puntos de extensión que permiten a esos hitos avanzar en paralelo sin tocar los mismos ficheros. Sin comportamiento nuevo: el host responde `unsupported` a todo lo que aún no maneja.

**Arquitectura:** sobre la rama `protocolo` (spec en 1356005, apilada sobre `m2.5a`).
- **Contrato:** `protocol/fixtures/{host,app}/<tipo>[@<variante>].json` y `protocol/fixtures/frames.json`. Go (`host/internal/bridge/fixtures_test.go`) y Kotlin (`FixturesTest.kt`) decodifican cada fixture en modo estricto y vuelven a codificar la base de cada tipo comparando el JSON. Cada lado tiene un registro `tipo → payload` por sentido, y un test lo cruza con los ficheros.
- **Host:** `bridge` gana los tipos por área (`protocol_chat.go`, `protocol_files.go`, `protocol_terminal.go` y `protocol_ui.go`), `registry.go`, `frame.go`, `rid` en el sobre, `Server.Send` (respuestas y flujos, nunca en el búfer) y `Options.OnFrame`. `cli/phone.go` cambia el `switch` por un registro de manejadores, con un fichero vacío por hito (`phone_chat.go`, `phone_agent.go` y `phone_ui.go`).
- **App:** los mismos tipos (`ProtocolChat.kt`, `ProtocolFiles.kt`, `ProtocolTerminal.kt` y `ProtocolUi.kt`), `Registry.kt` y `Frame.kt`. `BridgeState` delega en reductores por área (`ConfirmReducer`, `ChatReducer` y `RemoteReducer`), y `Bridge` gana `listen`, `listenFrames`, `send`, `request` y `sendFrame`.

**Tecnologías:** las de M2. Nada nuevo en `go.mod` ni en `libs.versions.toml`.

**Spec:** `docs/superpowers/specs/2026-10-01-adbuddy-protocolo-m3-m5-design.md` (todo). Se apoya en la spec de la v1 (§5.2-§5.7).

```go
package bridge

import (

// fixtureDir holds the protocol's shared fixtures (protocol spec §12), read
// by these tests and by the app's FixturesTest.
const fixtureDir = "../../../protocol/fixtures"

// strictDecode decodes data into v, failing on any field v does not have.
func strictDecode(data []byte, v any) error {
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.DisallowUnknownFields()
	return dec.Decode(v)
}

// sameJSON reports whether a and b are the same JSON value.
func sameJSON(t *testing.T, a, b []byte) bool {
	t.Helper()
	var x, y any
	if err := json.Unmarshal(a, &x); err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(b, &y); err != nil {
		t.Fatal(err)
	}
	return reflect.DeepEqual(x, y)
}

// fixtures returns the fixture files of a direction (host or app) by name:
// "<type>" for the base file, "<type>@<variant>" for the others.
func fixtures(t *testing.T, dir string) map[string]string {
	t.Helper()
	files, err := filepath.Glob(filepath.Join(fixtureDir, dir, "*.json"))
	if err != nil {
		t.Fatal(err)
	}
	if len(files) == 0 {
		t.Fatalf("no fixtures in %s", filepath.Join(fixtureDir, dir))
	}
	out := map[string]string{}
	for _, f := range files {
		out[strings.TrimSuffix(filepath.Base(f), ".json")] = f
	}
	return out
}

// Each fixture decodes strictly into its type's payload, and the base file
// of each type comes out the same when encoded again: a field renamed,
// added or dropped on either side breaks this test or the app's.
func TestFixtures(t *testing.T) {
	for dir, payloads := range map[string]map[string]func() any{"host": HostPayloads, "app": AppPayloads} {
		for name, path := range fixtures(t, dir) {
			t.Run(dir+"/"+name, func(t *testing.T) {
				data, err := os.ReadFile(path)
				if err != nil {
					t.Fatal(err)
				}
				var m Message
				if err := strictDecode(data, &m); err != nil {
					t.Fatalf("envelope: %v", err)
				}
				typ, variant, _ := strings.Cut(name, "@")
				if m.Type != typ || m.V != ProtocolVersion {
					t.Fatalf("type %q v%d in %s", m.Type, m.V, name)
				}
				if len(m.RID) > MaxRID {
					t.Fatalf("rid of %d characters", len(m.RID))
				}
				newPayload, ok := payloads[typ]
				if !ok {
					t.Fatalf("%s is not a type the %s sends", typ, dir)
				}
				v := newPayload()
				if err := strictDecode(m.Payload, v); err != nil {
					t.Fatalf("payload: %v", err)
				}
				if variant != "" {
					return
				}
				again, err := json.Marshal(v)
				if err != nil {
					t.Fatal(err)
				}
				if !sameJSON(t, again, m.Payload) {
					t.Fatalf("encoded again:\n%s\nfixture:\n%s", again, m.Payload)
				}
			})
		}
	}
}

// Every type has a base fixture, and every fixture a type.
func TestEveryTypeHasAFixture(t *testing.T) {
	for dir, payloads := range map[string]map[string]func() any{"host": HostPayloads, "app": AppPayloads} {
		files := fixtures(t, dir)
		var missing []string
		for typ := range payloads {

	}

type frameVectors struct {
	Valid []struct {
		ID    string `json:"id"`
		Data  string `json:"data"` // base64
		Frame string `json:"frame"`
	} `json:"valid"`
	Invalid []struct {
		Why   string `json:"why"`
		Frame string `json:"frame"`
	} `json:"invalid"`
	BadIDs []string `json:"badIds"`
}

func TestFrameVectors(t *testing.T) {
	data, err := os.ReadFile(filepath.Join(fixtureDir, "frames.json"))
	if err != nil {
		t.Fatal(err)
	}
	var v frameVectors
	if err := strictDecode(data, &v); err != nil {
		t.Fatal(err)
	}
	for _, c := range v.Valid {
		id, err := ParseFrameID(c.ID)
		if err != nil {
			t.Fatal(err)
		}
		payload, _ := base64.StdEncoding.DecodeString(c.Data)
		frame, err := EncodeFrame(id, payload)
		if err != nil || hex.EncodeToString(frame) != c.Frame {
			t.Errorf("EncodeFrame(%s, %q) = %x, %v; want %s", c.ID, payload, frame, err, c.Frame)
		}
		raw, _ := hex.DecodeString(c.Frame)
		gotID, gotData, err := DecodeFrame(raw)
		if err != nil || gotID.String() != c.ID || !bytes.Equal(gotData, payload) {
			t.Errorf("DecodeFrame(%s) = %s %q %v", c.Frame, gotID, gotData, err)
		}
	}
	for _, c := range v.Invalid {
		raw, _ := hex.DecodeString(c.Frame)
		if _, _, err := DecodeFrame(raw); err == nil {
			t.Errorf("DecodeFrame accepted a frame with %s", c.Why)
		}
	}
	for _, s := range v.BadIDs {
		if _, err := ParseFrameID(s); err == nil {
			t.Errorf("ParseFrameID(%q) accepted it", s)
		}
	}
}

func TestFrameSizeLimits(t *testing.T) {
	var id FrameID
	if _, err := EncodeFrame(id, nil); err == nil {
		t.Error("a frame without data was encoded")
	}
	if _, err := EncodeFrame(id, make([]byte, MaxFrameData+1)); err == nil {
		t.Error("a frame over 256 KiB was encoded")
	}
	frame, err := EncodeFrame(id, make([]byte, MaxFrameData))
	if err != nil {
		t.Fatal(err)
	}
	if _, _, err := DecodeFrame(frame); err != nil {
		t.Errorf("a 256 KiB frame was refused: %v", err)
	}
	if _, _, err := DecodeFrame(append(frame, 0)); err == nil {
		t.Error("a frame over 256 KiB was decoded")
	}
}
```

```go
package bridge

import (
	"encoding/hex"
	"errors"
	"fmt"
)

// Binary frames carry the bytes of uploads and downloads (protocol spec
// §6.4): the 16-byte id of the attachment followed by 1 B to 256 KiB of data.
const (
	FrameIDLen   = 16
	MaxFrameData = 256 << 10
)

// FrameID is an attachment's id as its frames carry it. In JSON it is 32
// lower-case hex digits (Attachment.ID).
type FrameID [FrameIDLen]byte

func (id FrameID) String() string { return hex.EncodeToString(id[:]) }

// ParseFrameID reads an id written as 32 lower-case hex digits.
func ParseFrameID(s string) (FrameID, error) {
	var id FrameID
	if len(s) != 2*FrameIDLen {
		return id, fmt.Errorf("frame id %q: want %d hex digits", s, 2*FrameIDLen)
	}
	for _, c := range s {
		if !('0' <= c && c <= '9' || 'a' <= c && c <= 'f') {
			return id, fmt.Errorf("frame id %q: want lower-case hex digits", s)
		}
	}
	_, err := hex.Decode(id[:], []byte(s))
	return id, err
}

// ErrFrameSize is a frame without data, or with more than MaxFrameData.
var ErrFrameSize = errors.New("frame data must be 1 B to 256 KiB")

// EncodeFrame returns the binary frame of data for id.
func EncodeFrame(id FrameID, data []byte) ([]byte, error) {
	if len(data) == 0 || len(data) > MaxFrameData {
		return nil, ErrFrameSize
	}
	return append(id[:], data...), nil
}

// DecodeFrame splits a binary frame into its id and its data. The data shares
// the frame's memory.
func DecodeFrame(frame []byte) (FrameID, []byte, error) {
	var id FrameID
	if len(frame) <= FrameIDLen || len(frame) > FrameIDLen+MaxFrameData {
		return id, nil, ErrFrameSize
	}
	copy(id[:], frame)
	return id, frame[FrameIDLen:], nil
}
```

```kotlin
package tech.hybriq.adbuddy.bridge

/**
 * Binary frames (protocol spec §6.4): the 16-byte id of an attachment followed
 * by 1 B to 256 KiB of its data. Mirrors host/internal/bridge/frame.go.
 */
object Frame {
    const val ID_LEN = 16
    const val MAX_DATA = 256 * 1024

    /** The id written as 32 lower-case hex digits, or null when it is not. */
    fun parseId(s: String): ByteArray? {
        if (s.length != 2 * ID_LEN || s.any { it !in '0'..'9' && it !in 'a'..'f' }) return null
        return ByteArray(ID_LEN) { i -> s.substring(2 * i, 2 * i + 2).toInt(16).toByte() }
    }

    fun idString(id: ByteArray): String = id.joinToString("") { "%02x".format(it) }

    /** The frame of data for id, or null for a bad id or a size outside 1 B to 256 KiB. */
    fun encode(id: String, data: ByteArray): ByteArray? {
        val raw = parseId(id) ?: return null
        if (data.isEmpty() || data.size > MAX_DATA) return null
        return raw + data
    }

    /** The id and the data of a frame, or null when its size is wrong. */
    fun decode(frame: ByteArray): Pair<String, ByteArray>? {
        if (frame.size <= ID_LEN || frame.size > ID_LEN + MAX_DATA) return null
        return idString(frame.copyOfRange(0, ID_LEN)) to frame.copyOfRange(ID_LEN, frame.size)
    }
}
```

`ConfirmReducer.kt` (la lógica de M2 de `BridgeState.apply` y `snapshot`, sin cambios de comportamiento):

```kotlin
package tech.hybriq.adbuddy.bridge

/** M2: the session's state, the 🔴 questions and the journal. */
object ConfirmReducer : Reducer {
    override val types = setOf(Protocol.SESSION_STATE, Protocol.CONFIRM_REQUEST, Protocol.JOURNAL_ENTRY)

    override fun event(s: BridgeState, m: Message): BridgeState = when (m.type) {
        Protocol.SESSION_STATE -> s.copy(sessionState = Protocol.payload<SessionState>(m) ?: s.sessionState)
        Protocol.CONFIRM_REQUEST -> {
            val r = Protocol.payload<ConfirmRequest>(m)
            if (r == null) s else s.copy(pending = s.pending.filter { it.id != r.id } + r)
        }
        Protocol.JOURNAL_ENTRY -> {
            val e = Protocol.payload<JournalEntry>(m)
            if (e == null) {
                s
            } else {
                // A final record of a pending question means it was answered (here, on the PC or by expiry).
                val final = e.status != "running"
                val stillPending = if (final) s.pending.filter { it.id != e.id } else s.pending
                // A newer record of the same command takes its place in the list.
                val i = s.journal.indexOfFirst { it.id == e.id }
                val updated = if (i >= 0) s.journal.toMutableList().also { it[i] = e } else s.journal + e
                s.copy(
                    pending = stillPending, journal = updated.takeLast(BridgeState.MAX_JOURNAL),
                    answering = if (final) s.answering - e.id else s.answering,
                )
            }
        }
        else -> s
    }

    override fun snapshot(s: BridgeState, snap: Snapshot): BridgeState {
        val entries = snap.journal.mapNotNull {
            try {
                Protocol.json.decodeFromJsonElement(JournalEntry.serializer(), it)
            } catch (e: IllegalArgumentException) {
                null
            } catch (e: StackOverflowError) {
                null
            }
        }
        val ids = snap.pending.map { it.id }.toSet()
        return s.copy(
            sessionState = snap.state, pending = snap.pending, journal = entries.takeLast(BridgeState.MAX_JOURNAL),
            answering = s.answering.filter { it in ids }.toSet(),
        )
    }
}
```
