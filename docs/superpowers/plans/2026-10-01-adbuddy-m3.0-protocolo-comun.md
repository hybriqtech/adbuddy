# ADBuddy M3.0 · Protocolo común · Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** dejar escrito y probado a los dos lados el contrato completo del protocolo app ↔ host de M3, M4 y M5 (tipos, fixtures compartidas, clases de mensaje, `rid` y `error`, tramas binarias) y los puntos de extensión que permiten a esos hitos avanzar en paralelo sin tocar los mismos ficheros. Sin comportamiento nuevo: el host responde `unsupported` a todo lo que aún no maneja.

**Arquitectura:** sobre la rama `protocolo` (spec corregida, último commit b7104ab, apilada sobre `m2.5a`).
- **Contrato:** `protocol/fixtures/{host,app}/<tipo>[@<variante>].json` y `protocol/fixtures/frames.json`.
  - Go (`host/internal/bridge/fixtures_test.go`) y Kotlin (`FixturesTest.kt`) decodifican cada fixture en modo estricto y vuelven a codificar la base de cada tipo comparando el JSON.
  - Comprueban que cada campo de cada tipo aparece en alguna fixture.
  - Cada lado tiene un registro `tipo → payload` por sentido, cruzado con los ficheros y con las constantes de tipo.
  - Las fixtures del host llevan `seq` y `rid` según su clase.
- **Host:** `bridge` gana:
  - los tipos por área (`protocol_chat.go`, `protocol_files.go`, `protocol_terminal.go` y `protocol_ui.go`), `registry.go` (con `HostClasses`) y `frame.go` (id, offset y datos);
  - `rid` en el sobre;
  - un método por clase de mensaje: `Publish`/`PublishRID` para eventos, `Reply` para respuestas (atada a la conexión que preguntó) y `Send` para flujos; cada uno rechaza las demás clases;
  - `Options.OnFrame` y `Options.OnAttach`.

  `cli/phone.go` cambia el `switch` por un `area` por hito (`phone_areas.go`), con un fichero vacío para cada uno (`phone_chat.go`, `phone_agent.go` y `phone_ui.go`).
- **App:** los mismos tipos (`ProtocolChat.kt`, `ProtocolFiles.kt`, `ProtocolTerminal.kt` y `ProtocolUi.kt`), `Registry.kt` y `Frame.kt`. `BridgeState` delega en reductores por área (`ConfirmReducer`, `ChatReducer` y `RemoteReducer`). `Bridge` gana:
  - `listen` y `listenFrames`, con varios oyentes;
  - `send`;
  - `request(tipo, payload, onResult)`, con pendientes que fallan al caer la conexión;
  - `sendFrame` y `queuedBytes`.

**Tecnologías:** las de M2. Nada nuevo en `go.mod` ni en `libs.versions.toml`.

**Spec:** `docs/superpowers/specs/2026-10-01-adbuddy-protocolo-m3-m5-design.md` (todo). Se apoya en la spec de la v1 (§5.2-§5.7).

```go
// fixtureDir holds the protocol's shared fixtures (protocol spec §12), read
// by these tests and by the app's FixturesTest.
const fixtureDir = "../../../protocol/fixtures"

var directions = map[string]map[string]func() any{"host": HostPayloads, "app": AppPayloads}

// strictDecode decodes data into v, failing on any field v does not have.
func strictDecode(data []byte, v any) error {
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.DisallowUnknownFields()
	return dec.Decode(v)
}

// sameJSON reports whether a and b are the same JSON value.
func sameJSON(t *testing.T, a, b []byte) bool {
	t.Helper()
	var x, y any
	if err := json.Unmarshal(a, &x); err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(b, &y); err != nil {
		t.Fatal(err)
	}
	return reflect.DeepEqual(x, y)
}

// fixtures returns the fixture files of a direction (host or app) by name:
// "<type>" for the base file, "<type>@<variant>" for the others.
func fixtures(t *testing.T, dir string) map[string]string {
	t.Helper()
	files, err := filepath.Glob(filepath.Join(fixtureDir, dir, "*.json"))
	if err != nil {
		t.Fatal(err)
	}
	if len(files) == 0 {
		t.Fatalf("no fixtures in %s", filepath.Join(fixtureDir, dir))
	}
	out := map[string]string{}
	for _, f := range files {
		out[strings.TrimSuffix(filepath.Base(f), ".json")] = f
	}
	return out
}

// readFixture reads a fixture's envelope strictly.
func readFixture(t *testing.T, path string) Message {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	var m Message
	if err := strictDecode(data, &m); err != nil {
		t.Fatalf("%s: envelope: %v", path, err)
	}
	return m
}

// Each fixture decodes strictly into its type's payload, and the base file
// of each type comes out the same when encoded again: a field renamed or
// dropped on either side breaks this test or the app's.
func TestFixtures(t *testing.T) {
	for dir, payloads := range directions {
		for name, path := range fixtures(t, dir) {
			t.Run(dir+"/"+name, func(t *testing.T) {
				m := readFixture(t, path)
				typ, variant, _ := strings.Cut(name, "@")
				if m.Type != typ || m.V != ProtocolVersion {
					t.Fatalf("type %q v%d in %s", m.Type, m.V, name)
				}
				if m.RID != "" && !ValidRID(m.RID) {
					t.Fatalf("rid %q", m.RID)
				}
				newPayload, ok := payloads[typ]
				if !ok {
					t.Fatalf("%s is not a type the %s sends", typ, dir)
				}
				v := newPayload()
				if err := strictDecode(m.Payload, v); err != nil {
					t.Fatalf("payload: %v", err)
				}
				if variant != "" {
					return
				}
				again, err := json.Marshal(v)
				if err != nil {
					t.Fatal(err)
				}
				if !sameJSON(t, again, m.Payload) {
					t.Fatalf("encoded again:\n%s\nfixture:\n%s", again, m.Payload)
				}
			})
		}
	}
}

// typePaths adds the JSON paths of t's fields ("attachment.thumb"): lists are
// transparent, and time.Time, byte slices (base64, json.RawMessage) and maps
// are leaves.
func typePaths(t reflect.Type, prefix string, out map[string]bool) {
	for t.Kind() == reflect.Pointer || t.Kind() == reflect.Slice && t.Elem().Kind() != reflect.Uint8 {
		t = t.Elem()
	}
	if t.Kind() != reflect.Struct || t == reflect.TypeOf(time.Time{}) {
		return
	}
	for i := range t.NumField() {
		f := t.Field(i)
		name, _, _ := strings.Cut(f.Tag.Get("json"), ",")
		if !f.IsExported() || name == "-" {
			continue
		}
		if name == "" {
			name = f.Name
		}
		out[prefix+name] = true
		typePaths(f.Type, prefix+name+".", out)
	}
}

// jsonPaths adds the paths of the keys in v, a decoded JSON value.
func jsonPaths(v any, prefix string, out map[string]bool) {
	switch v := v.(type) {
	case map[string]any:
		for k, x := range v {
			out[prefix+k] = true
			jsonPaths(x, prefix+k+".", out)
		}
	case []any:
		for _, x := range v {
			jsonPaths(x, prefix, out)
		}
	}
}

// Every field of every payload appears in at least one fixture of its type:
// a field added to one side only would otherwise go unseen, since an absent
// optional field is encoded again as absent.
func TestFixturesHaveEveryField(t *testing.T) {
	for dir, payloads := range directions {
		present := map[string]map[string]bool{}
		for name, path := range fixtures(t, dir) {
			typ, _, _ := strings.Cut(name, "@")
			if present[typ] == nil {
				present[typ] = map[string]bool{}
			}
			var v any
			if err := json.Unmarshal(readFixture(t, path).Payload, &v); err != nil {
				t.Fatal(err)
			}
			jsonPaths(v, "", present[typ])
		}
		for typ, newPayload := range payloads {
			want := map[string]bool{}
			typePaths(reflect.TypeOf(newPayload()), "", want)
```

```go
package bridge

import (
	"encoding/binary"
	"encoding/hex"
	"errors"
	"fmt"
	"math"
)

// Binary frames carry the bytes of uploads and downloads (protocol spec
// §6.4): the 16-byte id of the attachment, the 8-byte big-endian offset of
// the data in the file, and 1 B to 256 KiB of data. The offset lets the
// receiver drop frames that do not follow what it has (a cancelled download,
// a resumed upload).
const (
	FrameIDLen     = 16
	FrameOffsetLen = 8
	FrameHeaderLen = FrameIDLen + FrameOffsetLen
	MaxFrameData   = 256 << 10
)

// FrameID is an attachment's id as its frames carry it. In JSON it is 32
// lower-case hex digits (Attachment.ID).
type FrameID [FrameIDLen]byte

func (id FrameID) String() string { return hex.EncodeToString(id[:]) }

// ParseFrameID reads an id written as 32 lower-case hex digits.
func ParseFrameID(s string) (FrameID, error) {
	var id FrameID
	if len(s) != 2*FrameIDLen {
		return id, fmt.Errorf("frame id %q: want %d hex digits", s, 2*FrameIDLen)
	}
	for _, c := range s {
		if !('0' <= c && c <= '9' || 'a' <= c && c <= 'f') {
			return id, fmt.Errorf("frame id %q: want lower-case hex digits", s)
		}
	}
	_, err := hex.Decode(id[:], []byte(s))
	return id, err
}

// ErrFrame is a frame without data, with more than MaxFrameData, or with a
// negative offset.
var ErrFrame = errors.New("a frame has an id, an offset from 0 and 1 B to 256 KiB of data")

// EncodeFrame returns the binary frame of data at offset in the file id.
func EncodeFrame(id FrameID, offset int64, data []byte) ([]byte, error) {
	if len(data) == 0 || len(data) > MaxFrameData || offset < 0 {
		return nil, ErrFrame
	}
	frame := make([]byte, FrameHeaderLen, FrameHeaderLen+len(data))
	copy(frame, id[:])
	binary.BigEndian.PutUint64(frame[FrameIDLen:], uint64(offset))
	return append(frame, data...), nil
}

// DecodeFrame splits a binary frame into its id, its offset and its data. The
// data shares the frame's memory.
func DecodeFrame(frame []byte) (FrameID, int64, []byte, error) {
	var id FrameID
	if len(frame) <= FrameHeaderLen || len(frame) > FrameHeaderLen+MaxFrameData {
		return id, 0, nil, ErrFrame
	}
	off := binary.BigEndian.Uint64(frame[FrameIDLen:FrameHeaderLen])
	if off > math.MaxInt64 {
		return id, 0, nil, ErrFrame
	}
	copy(id[:], frame)
	return id, int64(off), frame[FrameHeaderLen:], nil
}
```

```go
// one builds its own in its own file (phone_chat.go, phone_agent.go,
// phone_ui.go), so lines of work in parallel do not edit the same code. A nil
// function does nothing.
type area struct {
	// handlers handle the app's message types; no two areas share one.
	handlers map[string]func(bridge.Message)
	// snapshot fills the area's part of a snapshot. It is called with the
	// server's lock held: no Publish, Send, Reply nor Connected.
	snapshot func(*bridge.Snapshot)
	// attach: an app authenticated, new or replacing another. What the area
	// keeps per connection (a terminal subscription, uploads) starts over.
	// A message of the replaced connection may still reach a handler after
	// this: a handler that keeps state per connection checks
	// l.srv.Current(m) first.
	attach func()
	// gone: no app is connected any more.
	gone func()
	// frame receives a binary frame from the app (uploads, M3).
	frame func(id bridge.FrameID, offset int64, data []byte)
}

// setAreas keeps the areas and merges their handlers. A type in two areas is
// a programming error.
func (l *phoneLink) setAreas(areas ...area) {
	l.areas = areas
	l.handlers = map[string]func(bridge.Message){}
	for _, a := range areas {
		for typ, h := range a.handlers {
			if _, dup := l.handlers[typ]; dup {
				panic(fmt.Sprintf("phone link: %s is handled by two areas", typ))
			}
			l.handlers[typ] = h
		}
	}
}

	for _, a := range l.areas {
```

`Frame.kt`:

```kotlin
package tech.hybriq.adbuddy.bridge

import java.nio.ByteBuffer

/** One binary frame: data at offset in the attachment id. */
class Chunk(val id: String, val offset: Long, val data: ByteArray)

/**
 * Binary frames (protocol spec §6.4): the 16-byte id of an attachment, the
 * 8-byte big-endian offset of the data in the file, and 1 B to 256 KiB of
 * data. Mirrors host/internal/bridge/frame.go.
 */
object Frame {
    const val ID_LEN = 16
    const val OFFSET_LEN = 8
    const val HEADER_LEN = ID_LEN + OFFSET_LEN
    const val MAX_DATA = 256 * 1024

    /** The id written as 32 lower-case hex digits, or null when it is not. */
    fun parseId(s: String): ByteArray? {
        if (s.length != 2 * ID_LEN || s.any { it !in '0'..'9' && it !in 'a'..'f' }) return null
        return ByteArray(ID_LEN) { i -> s.substring(2 * i, 2 * i + 2).toInt(16).toByte() }
    }

    fun idString(id: ByteArray): String = id.joinToString("") { "%02x".format(it) }

    /** The frame of data at offset in id, or null for a bad id, a negative offset or a size outside 1 B to 256 KiB. */
    fun encode(id: String, offset: Long, data: ByteArray): ByteArray? {
        val raw = parseId(id) ?: return null
        if (offset < 0 || data.isEmpty() || data.size > MAX_DATA) return null
        return ByteBuffer.allocate(HEADER_LEN + data.size).put(raw).putLong(offset).put(data).array()
    }

    /** The chunk a frame carries, or null when its size or its offset is wrong. */
    fun decode(frame: ByteArray): Chunk? {
        if (frame.size <= HEADER_LEN || frame.size > HEADER_LEN + MAX_DATA) return null
        val offset = ByteBuffer.wrap(frame, ID_LEN, OFFSET_LEN).long
        if (offset < 0) return null
        return Chunk(idString(frame.copyOfRange(0, ID_LEN)), offset, frame.copyOfRange(HEADER_LEN, frame.size))
    }
}
```

`ConfirmReducer.kt` (la lógica de M2 de `BridgeState.apply` y `snapshot`, sin cambios de comportamiento):

```kotlin
package tech.hybriq.adbuddy.bridge

/** M2: the session's state, the 🔴 questions and the journal. */
object ConfirmReducer : Reducer {
    override val types = setOf(Protocol.SESSION_STATE, Protocol.CONFIRM_REQUEST, Protocol.JOURNAL_ENTRY)

    override fun event(s: BridgeState, m: Message): BridgeState = when (m.type) {
        Protocol.SESSION_STATE -> s.copy(sessionState = Protocol.payload<SessionState>(m) ?: s.sessionState)
        Protocol.CONFIRM_REQUEST -> {
            val r = Protocol.payload<ConfirmRequest>(m)
            if (r == null) s else s.copy(pending = s.pending.filter { it.id != r.id } + r)
        }
        Protocol.JOURNAL_ENTRY -> {
            val e = Protocol.payload<JournalEntry>(m)
            if (e == null) {
                s
            } else {
                // A final record of a pending question means it was answered (here, on the PC or by expiry).
                val final = e.status != "running"
                val stillPending = if (final) s.pending.filter { it.id != e.id } else s.pending
                // A newer record of the same command takes its place in the list.
                val i = s.journal.indexOfFirst { it.id == e.id }
                val updated = if (i >= 0) s.journal.toMutableList().also { it[i] = e } else s.journal + e
                s.copy(
                    pending = stillPending, journal = updated.takeLast(BridgeState.MAX_JOURNAL),
                    answering = if (final) s.answering - e.id else s.answering,
                )
            }
        }
        else -> s
    }

    override fun snapshot(s: BridgeState, snap: Snapshot): BridgeState {
        val entries = snap.journal.mapNotNull {
            try {
                Protocol.json.decodeFromJsonElement(JournalEntry.serializer(), it)
            } catch (e: IllegalArgumentException) {
                null
            } catch (e: StackOverflowError) {
                null
            }
        }
        val ids = snap.pending.map { it.id }.toSet()
        return s.copy(
            sessionState = snap.state, pending = snap.pending, journal = entries.takeLast(BridgeState.MAX_JOURNAL),
            answering = s.answering.filter { it in ids }.toSet(),
        )
    }
}
```
